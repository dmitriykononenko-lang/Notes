## Окружение и подготовка

1. Установить актуальный архив `task-templates.zip` в интеграцию
   (amoМаркет → Разработчикам → интеграция «Шаблоны задач» → загрузить новый
   архив → Сохранить).
2. Жёстко обновить страницу: **Ctrl+Shift+R**.
3. Тест проводить минимум под двумя ролями: **администратор** и **обычный
   пользователь** (без прав администратора) — где это указано в кейсе.

Отметки: ✅ прошёл / ❌ не прошёл (с описанием) / ⏭️ пропущен.

## Приложение А. Диагностика пункта в переключателе (для кейса 2.3)

Открыть сделку → раскрыть переключатель внизу карточки → F12 → Console →
вставить и приложить вывод:

```js
(function () {
  var LT = ['Задача','Task'], LN = ['Примечание','Note'];
  function txt(el){ return (el.textContent||'').replace(/\s+/g,' ').trim(); }
  var out = [];
  out.push('styles injected: ' + !!document.getElementById('yp-tt-styles'));
  out.push('our item present: ' + document.querySelectorAll('.yp-tt-compose-item').length);
  var leaves = [].slice.call(document.querySelectorAll('*')).filter(function(el){
    return el.getElementsByTagName('*').length <= 20 && LT.indexOf(txt(el)) !== -1;
  });
  out.push('leaf "Задача" candidates: ' + leaves.length);
  leaves.slice(0,3).forEach(function(leaf, i){
    var node = leaf, menu = null;
    for (var lvl=0; lvl<5 && node; lvl++){
      var kids = node.children || [];
      if (kids.length>=2 && kids.length<=8 && [].some.call(kids,function(k){return LN.indexOf(txt(k))!==-1;})){ menu = node; break; }
      node = node.parentNode;
    }
    out.push('--- cand '+i+' menu found: '+!!menu);
    out.push(menu ? menu.outerHTML.slice(0,700) : 'parent: ' + (leaf.parentNode?leaf.parentNode.outerHTML.slice(0,500):'none'));
  });
  console.log(out.join('\n'));
})();
```
